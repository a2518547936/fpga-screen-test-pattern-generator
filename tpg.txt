        h_count_full    <= 1'b0;
        v_count_full    <= 1'b0;
    end
    else begin
        h_count_full    <= (h_count == HS_total - 1) ? 1'b1:1'b0;
        v_count_full    <= (v_count == VS_total) ? 1'b1:1'b0;
    end
end

always @(posedge clk or negedge rstn) begin
    if(!rstn)
    begin
        h_count <=  16'd0;
    end
    else if(h_count_full)
    begin
        h_count <=  16'd1;
    end
    else
    begin
        h_count <=  h_count + 16'd1;
    end
end

always @(posedge clk or negedge rstn)
begin
    if(!rstn)
    begin
        v_count <=  16'd0;
    end
    else if(h_count_full)
    begin
        if(v_count_full)
        begin
            v_count <=  16'd1;
        end
        else
        begin
            v_count  <=  v_count + 16'd1;
        end
    end
    else
    begin
        v_count <=  v_count;
    end
end

        HS_r    <=  1'b0;

    begin
        HS_r    <=  HS_r;
    end

    if(!rstn)
    begin
        VS_r    <=  1'b0;
    end
    else if((v_count == VS_width) & (h_count_full))
    begin
        VS_r    <=  1'b1;
    end
    else if((v_count == VS_total) & (h_count_full))
    begin
        VS_r    <=  1'b0;
    end
    else
    begin
        VS_r    <=  VS_r;
    end
end

begin
    if(!rstn)
    begin
        DE_h_r    <=  1'b0;
    end
    else if(h_count == HS_back)
    begin
        DE_h_r    <=  1'b1;
    end
    else if(h_count == HS_back_display)
    begin
        DE_h_r    <=  1'b0;
    end
end

always @(posedge clk or negedge rstn)
begin
    if(!rstn)
    begin
        DE_v_r    <=  1'b0;
    end
    else if(v_count > VS_width_back && v_count <= VS_width_back_display)
    begin
        DE_v_r    <=  1'b1;
    end
    else 
    begin
        DE_v_r    <=  1'b0;
    end
end

if(!rstn) begin
    DE_r    <=  1'b0;
end
else begin
    DE_r    <=  DE_v_r & DE_h_r;
